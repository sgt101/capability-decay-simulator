\noindent
The state is per-person latent expertise $E_i \in [0,1]$. Each tick runs
institution statistics, then the expertise update, then mobility, then turnover,
in that order. Capability is a read-out, never a state variable.

\section*{Conventions}
\begin{itemize}[leftmargin=1.4em,itemsep=1pt,topsep=3pt]
  \item $1$ tick $=1$ month; a career is $40$ years $=480$ ticks.
  \item Expertise and the AI level live on $[0,1]$.
  \item Expert threshold $\theta = 0.585$ (anchors $w(\theta)=1$ and \key{shareExpert}).
  \item Capability base $\rho = 1000$; move-selection temperature $\tau = 0.12$.
  \item Defaults quoted below are the monthly calibration \key{simulator.html}
        runs (\key{DEFAULT\_PARAMS} with \key{MONTHLY\_TICK\_PARAMS} applied). A
        second value, where given, is the raw dimensionless default, the
        entrant-pipeline overlay, or the world-model deployment value.
\end{itemize}

\section{Governing equations}

\subsection*{Setup ($t = 0$)}

\begin{equation}\label{eq:E1}
  G = \operatorname{BA}(M, m), \qquad
  P_j = \text{normalised degree of institution } j .
\end{equation}

\begin{equation}\label{eq:E2}
  E_i(0) \sim \operatorname{SkewNormal}(\mu_E,\, \omega_E,\, \alpha_E),
  \ \text{clipped to } [0,1].
\end{equation}

\begin{equation}\label{eq:E3}
  L_i = e^{\sigma_L Z}, \qquad Z \sim \mathcal{N}(0,1).
\end{equation}
\eqnote{The per-person rate multiplier: drawn once, fixed for the career. It
scales learning, decay \emph{and} mobility, so a fast learner also moves more.}

\begin{equation}\label{eq:E4}
  a_i \sim \mathcal{N}(\mu_a,\, \omega_a) \ \text{on } [0,1]
  \qquad (a_i = 1 \text{ when } \omega_a = 0).
\end{equation}

\begin{equation}\label{eq:E5}
  A = \lambda \cdot \max_i E_i(0).
\end{equation}
\eqnote{The absolute AI level. Defined only when AI is enabled; fixed at $t=0$, so
it never chases the live top performer.}

\subsection*{Each tick --- institution statistics}

\begin{equation}\label{eq:E6}
  \bar E_j = \operatorname{mean}\{E_i : i \in j\}.
\end{equation}

\begin{equation}\label{eq:E7}
  T_j = \operatorname{mean}\bigl(\text{top-}n_t \text{ by } E \text{ in } j,\
        \text{tenure} \ge T_s \text{ yr}\bigr).
\end{equation}
\eqnote{Teaching level. Falls back to $\bar E_j$ when $T_s = 0$ or $j$ has no
member past the tenure bar.}

\begin{equation}\label{eq:E8}
  \varphi_j = \frac{x_j^{\,h}}{x_j^{\,h} + n_0^{\,h}}, \qquad
  x_j = \#\{\text{experts in } j\}.
\end{equation}
\eqnote{Critical-mass transfer efficiency, a Hill function of the expert count.
$\varphi_j = 1$ everywhere when $n_0 = 0$.}

\subsection*{Each tick --- expertise update (person $i$ in institution $j$)}

\begin{equation}\label{eq:E9}
  \operatorname{gap} = \min(a_i,\, T_j) - E_i, \qquad
  \operatorname{below} = \bigl[\, \text{AI enabled} \wedge E_i < A \,\bigr].
\end{equation}

\begin{equation}\label{eq:E10}
  \operatorname{gap} > 0 \ \text{(learning)}: \qquad
  \Delta E_i = \beta \, \operatorname{gap} L_i \cdot
    \bigl( \operatorname{below} \ ? \ \gamma_{\text{below}} : \gamma_{\text{above}} \bigr).
\end{equation}
\eqnote{The bracketed AI factor is applied only when AI is enabled; with AI off,
$\Delta E_i = \beta\,\operatorname{gap} L_i$. Three modifiers are suppressed above
for legibility, applied in this order \emph{before} the AI factor: the taught term
is capped at $c\,L_i$, then divided by $1 + k_d \max(0,\, E_i - \bar E)$ ($\bar E$
the live population mean), then multiplied by $\varphi_j$ (\eqn{E8}). The first two
are active in the entrant-pipeline calibration; $\varphi_j = 1$ unless
\key{criticalMass} is set.}

\begin{equation}\label{eq:E11}
  \begin{aligned}
    \operatorname{gap} \le 0 \ \text{(decay)}: \quad
    \Delta E_i &= \delta \, \operatorname{gap} L_i \;+\; \operatorname{ambient} \cdot
      \bigl( \operatorname{below} \ ? \ \gamma_{\text{below}} : \gamma_{\text{above}} \bigr), \\
    \operatorname{ambient} &= \mu_p \, \bar E_j^{\,0} \, L_i \max(0,\, a_i - E_i).
  \end{aligned}
\end{equation}
\eqnote{The $\delta\,\operatorname{gap} L_i$ term is never AI-gated --- the
bracketed factor multiplies \textup{ambient} only, and again only when AI is
enabled. $\bar E_j^{\,0}$ is the institution's \emph{founding} mean, fixed at
$t = 0$; $\mu_p$ may be negative.}

\begin{equation}\label{eq:E12}
  E_i \leftarrow \operatorname{clip}_{[0,1]}\!\bigl(E_i + \Delta E_i\bigr).
\end{equation}

\subsection*{Each tick --- mobility}

\begin{equation}\label{eq:E13}
  \Pr(\text{reconsider a move}) = p_{\text{move}}\, L_i .
\end{equation}
\eqnote{Candidate set $\mathcal{C}_i$: graph neighbours of the current
institution, or --- hybrid mode, with probability $\varepsilon$ --- up to $25$
institutions from anywhere. The current institution is always in
$\mathcal{C}_i$.}

\begin{equation}\label{eq:E14}
  U_{ij} = (1-\kappa)\max(0,\, \bar E_j - E_i)
         + \kappa\,(E_i - \bar E_j)
         + \eta\, P_j .
\end{equation}

\begin{equation}\label{eq:E15}
  \Pr(i \to j) = \frac{\exp(U_{ij}/\tau)}
                      {\sum_{k \in \mathcal{C}_i} \exp(U_{ik}/\tau)},
  \qquad \tau = 0.12 .
\end{equation}

\subsection*{Each tick --- turnover}

\begin{equation}\label{eq:E16}
  \text{retire } i \text{ w.p. } r; \qquad
  \text{refill w.p. } 1 \text{ or } \texttt{recruitmentFraction}.
\end{equation}
\eqnote{The freeze runs while
$\texttt{shockStart} \le t < \texttt{shockStart} + 12\,\texttt{shockYears}$; there
is no catch-up hiring afterwards.}

\begin{equation}\label{eq:E17}
  E_i = \max\!\bigl(E_{\min},\
        \operatorname{SkewNormal}(\mu_{\text{ent}},\, \omega_{\text{ent}},\, 0)
        \big|_{[0,1]}\bigr).
\end{equation}
\eqnote{A new entrant also redraws its rate multiplier $L_i$ (\eqn{E3}) and
ceiling $a_i$ (\eqn{E4}).}

\subsection*{Capability metric --- diagnostic, never feeds back into $E$}

\begin{equation}\label{eq:E18}
  w(E) = \rho^{\frac{E - \theta}{1 - \theta}}, \qquad
  \rho = 1000,\ \theta = 0.585
  \;\Longrightarrow\; w(\theta) = 1,\ w(1) = \rho .
\end{equation}
\eqnote{Per-person value weight, in threshold-expert equivalents. Convex: one
person at $E = 1$ is worth $\rho$ experts.}

\begin{equation}\label{eq:E19}
  \begin{aligned}
    \ell(E) &= F\,\alpha(E) + (1 - F)\,\beta_A(E), \qquad
              F = A^{\,b}, \qquad
              m(E) = \frac{1}{1 + e^{\,s(E - A)}},\\[2pt]
    \alpha(E) &= 1 + g_x + (g_n - g_x)\,m(E), \qquad
    \beta_A(E) = 1 - d_n\,m(E).
  \end{aligned}
\end{equation}
\eqnote{Asymmetric cognitive leverage (Dell'Acqua et al.): AI multiplies what a
person is worth, more inside the frontier $F$ than outside it. $\ell(E) \equiv 1$
whenever AI is disabled.}

\begin{equation}\label{eq:E20}
  C = \sum_i w(E_i)\,\ell(E_i), \qquad
  C_{\text{human}} = \sum_i w(E_i).
\end{equation}
